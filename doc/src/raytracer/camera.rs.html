48    fn initialize(&mut self) {
49        self.image_height = (self.image_width / self.aspect_ratio).max(1.0);
50
51        let theta = degrees_to_radians(self.fov);
52        let h = (theta / 2.0).tan();
53
54        let base_viewport_height = 2.0 * h;
55        let base_viewport_width = base_viewport_height * (self.image_width / self.image_height);
56
57        let w = (self.lookfrom - self.lookat).normalize();
58        let u = self.vup.cross(&w).normalize();
59        let v = w.cross(&u);
60
61        self.center = self.lookfrom;
62
63        let viewport_u = base_viewport_width * self.focus_dist * u;
64        let viewport_v = -base_viewport_height * self.focus_dist * v;
65
66        self.pixel_delta_u = viewport_u / self.image_width;
67        self.pixel_delta_v = viewport_v / self.image_height;
68
69        let viewport_upper_left =
70            self.center - (self.focus_dist * w) - viewport_u / 2.0 - viewport_v / 2.0;
71
72        self.pixel00_loc = viewport_upper_left + 0.5 * (self.pixel_delta_u + self.pixel_delta_v);
73        self.pixel_sample_scale = 1.0 / self.samples_per_pixel;
74
75        let defocus_radius = self.focus_dist * degrees_to_radians(self.defocus_angle / 2.0).tan();
76        self.defocus_disc_u = u * defocus_radius;
77        self.defocus_disc_v = v * defocus_radius;
78    }

147    pub fn render(self: &Self, world: &dyn Surface) {
148        let remaining = Arc::new(AtomicI32::new(self.image_height as i32));
149        let total_height = self.image_height as i32;
150        let remaining_clone = Arc::clone(&remaining);
151        let progress_thread = thread::spawn(move || {
152            loop {
153                let r = remaining_clone.load(Ordering::Relaxed);
154                eprint!(
155                    "\rRendering scanlines: {} / {}  ",
156                    total_height as i32 - r,
157                    total_height
158                );
159                io::stderr().flush().unwrap();
160                if r == 0 {
161                    break;
162                }
163                thread::sleep(Duration::from_millis(50));
164            }
165        });
166
167        let iter = 0..self.image_height as i32;
168        #[cfg(feature = "threaded")]
169        let iter = iter.into_par_iter();
170
171        let pixels: Vec<Vec<f32>> = iter.map(|y| {
172            let remaining = Arc::clone(&remaining);
173            remaining.fetch_sub(1, Ordering::Relaxed);
174
175            let mut rng: SmallRng = rand::make_rng();
176            let mut row = Vec::with_capacity(self.image_width as usize * 3);
177            for x in 0..self.image_width as i32 {
178                let mut col = Color::new(0.0, 0.0, 0.0);
179                for _ in 0..self.samples_per_pixel as i32 {
180                    let ray = self.get_ray(x, y, &mut rng);
181                    col += self.ray_color(&ray, &world, 50, &mut rng);
182                }
183                col *= self.pixel_sample_scale;
184                row.push(col.x);
185                row.push(col.y);
186                row.push(col.z);
187            }
188            row
189        }).collect();
190        progress_thread.join().unwrap();
191
192        let input_img: Vec<f32> = pixels.into_iter().flatten().collect();
193        eprintln!();
194        #[cfg(feature = "denoise")]
195        eprint!("Denoising image... ");
196        #[cfg(feature = "denoise")]
197        let mut output_img = vec![0.0f32; input_img.len()];
198        #[cfg(feature = "denoise")]
199        let device = oidn::Device::new();
200        #[cfg(feature = "denoise")]
201        oidn::RayTracing::new(&device)
202            .srgb(true)
203            .image_dimensions(self.image_width as usize, self.image_height as usize)
204            .filter(&input_img[..], &mut output_img[..])
205            .expect("OIDN filter error");
206        #[cfg(feature = "denoise")]
207        if let Err(e) = device.get_error() { eprintln!("OIDN error: {}", e.1); }
208        #[cfg(feature = "denoise")]
209        eprint!("{}", "Done!".green());
210        #[cfg(feature = "denoise")]
211        eprintln!();
212        
213        #[cfg(not(feature = "denoise"))]
214        let output_img = input_img;
215
216
217        eprint!("Writing image to disk... ");
218
219        let mut buffer: RgbImage = ImageBuffer::new(self.image_width as u32, self.image_height as u32);
220        for (chunk, pixel) in output_img.chunks(3).zip(buffer.pixels_mut()) {
221            let r = (chunk[0].clamp(0.0, 1.0).sqrt() * 255.999) as u8;
222            let g = (chunk[1].clamp(0.0, 1.0).sqrt() * 255.999) as u8;
223            let b = (chunk[2].clamp(0.0, 1.0).sqrt() * 255.999) as u8;
224            *pixel = image::Rgb([r, g, b]);
225        }
226
227        match buffer.save("image.png") {
228            Err(e) => eprintln!("Error writing file: {}", e),
229            Ok(()) => println!("Done."),
230        };
231
232        eprint!("{}", "Done! \n".green());
233        eprint!("{}", "\nDone!\n".green());
234    }

236    pub fn get_ray(&self, x: i32, y: i32, rng: &mut SmallRng) -> Ray {
237        let offset = self.sample_square(rng);
238        let pixel_sample = self.pixel00_loc
239            + ((x as f32 + offset.x) * self.pixel_delta_u)
240            + ((y as f32 + offset.y) * self.pixel_delta_v);
241
242        let primary_direction = (pixel_sample - self.center).normalize();
243
244        let ray_origin = if self.defocus_angle <= 0.0 {
245            self.center
246        } else {
247            self.sample_defocus_disk(rng)
248        };
249
250        Ray::new_with_primary(ray_origin, pixel_sample - ray_origin, primary_direction)
251    }
252
253    fn sample_square(&self, rng: &mut SmallRng) -> Vector3<f32> {
254        Vector3::new(
255            random_f32(rng, -0.5, 0.5),
256            random_f32(rng, -0.5, 0.5),
257            0.0,
258        )
259    }
260
261    fn sample_defocus_disk(&self, rng: &mut SmallRng) -> Vector3<f32> {
262        let p = random_in_unit_disk(rng);
263        self.center + (p.x * self.defocus_disc_u) + (p.y * self.defocus_disc_v)
264    }

266    fn sample_skybox(&self, ray: &Ray) -> Color {
267        let dir = ray.primary_direction().normalize();
268        let theta = dir.z.atan2(dir.x);
269        let phi = dir.y.acos();
270
271        let u = (theta + PI) / (2.0 * PI);
272        let v = phi / PI;
273
274        self.skybox.sample(&u, &v, &dir)
275    }

277    fn ray_color(&self, ray: &Ray, world: &&dyn Surface, depth: i32, rng: &mut SmallRng) -> Color {
278        if depth <= 0 {
279            return Color::new(0.0, 0.0, 0.0);
280        }
281
282        let mut rec = HitRecord::default();
283        if !world.hit(ray, Interval::new(0.001, INFINITY), &mut rec) {
284            return self.sample_skybox(&ray);
285        }
286
287        let mut scattered = Ray::new(Vector3::zeros(), Vector3::new(1.0, 0.0, 0.0));
288        let mut attenuation = Color::new(0.0, 0.0, 0.0);
289        let col_emission = rec.mat.emitted(rec.u as f64, rec.v as f64, &rec.p);
290
291        if !rec.mat.scatter(ray, &rec, &mut attenuation, &mut scattered, rng) {
292            return col_emission;
293        }
294
295        let col_scatter = attenuation.component_mul(&self.ray_color(&scattered, world, depth - 1, rng));
296        return col_emission + col_scatter;
297    }
