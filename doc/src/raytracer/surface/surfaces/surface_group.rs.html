7pub struct SurfaceGroup {

9    bbox: AABB,

12impl SurfaceGroup {
13    pub fn new() -> Self {
14        Self {
15            objects: Vec::new(),
16            bbox: AABB::new_empty()
17        }
18    }
19
20    pub fn with(object: Arc<dyn Surface + Send + Sync>) -> Self {
21        let mut group = Self::new();
22        group.add(object);
23        group
24    }
25
26    pub fn add(&mut self, object: Arc<dyn Surface + Send + Sync>) {
27        self.bbox = AABB::new_from_boxes(&self.bbox, object.bounding_box());
28        self.objects.push(object);
29    }
30
31    pub fn into_objects(self) -> Vec<Arc<dyn Surface + Send + Sync>> {
32        self.objects
33    }

37    }

40impl Surface for SurfaceGroup {
41    fn hit(&self, ray: &Ray, ray_t: Interval, rec: &mut HitRecord) -> bool {
42        let mut temp_rec = HitRecord::default();
43
44        let mut hit_anything = false;
45        let mut closest_so_far = ray_t.max;
46
47        for object in &self.objects {
48            if object.hit(ray, Interval::new(ray_t.min, closest_so_far), &mut temp_rec) {
49                hit_anything = true;
50                closest_so_far = temp_rec.t;
51
52                *rec = temp_rec.clone();
53            }
54        }
55
56        hit_anything
57    }
58    
59    fn bounding_box(&self) -> &crate::aabb::AABB {
60        &self.bbox
